  redis:
    container_name: "{{ cookiecutter.docker_name_prefix }}-redis-prod"
    image: redis:7-alpine
    volumes:
      - redis_data:/data
    healthcheck:
      test: [ "CMD", "redis-cli", "ping" ]
      interval: 5s
      retries: 5
    networks:
      - network-default
    restart: unless-stopped

  celery_worker:
    build: .
    container_name: "{{ cookiecutter.docker_name_prefix }}-celery-worker-prod"
    # Bypass the web entrypoint: workers must not run migrations or
    # collectstatic -- those belong to the web service.
    entrypoint: []
    command: [ "celery", "-A", "core", "worker", "--loglevel=info" ]
    environment:
      DJANGO_SETTINGS_MODULE: config.settings.prod
      CDN_ROOT: /cdn
    volumes:
      - app_data:/cdn
    depends_on:
      db:
        condition: service_healthy
      redis:
        condition: service_healthy
    networks:
      - network-default
    restart: unless-stopped
    env_file:
      - .env
